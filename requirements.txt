# VisionIQ - shared Python dependencies
# Used by: local development, automated tests, Python Intelligence service,
# and the database migration job. All install from this ONE requirements.txt.
# .NET dependencies are managed separately by NuGet (.csproj files).
# Installing a library does not grant database privileges: keep runtime and
# migration images, identities, secrets, and database permissions separate.
# Versions below are not a lockfile; pin/lock before production deployment.

# --- Web API and configuration ---
# FastAPI: defines HTTP endpoints for the Python Intelligence service.
fastapi
# Uvicorn: runs the FastAPI application as an ASGI server.
uvicorn[standard]
# Pydantic Settings: validates environment variables and .env configuration.
pydantic-settings
# HTTPX: makes asynchronous or synchronous HTTP requests to external APIs.
httpx

# --- AI model providers ---
# Google Gen AI SDK: calls Gemini models through Vertex AI / Google APIs.
google-genai
# Anthropic SDK: integrates Claude when that provider is enabled.
anthropic

# --- PostgreSQL persistence and migrations ---
# SQLAlchemy: builds database connections and ORM/query abstractions.
sqlalchemy
# Alembic: runs controlled PostgreSQL schema migrations (migration job only).
alembic>=1.16,<2
# Psycopg 3: PostgreSQL driver used by application and migration code.
psycopg[binary]

# --- Analysis and retrieval ---
# NumPy: numerical operations and similarity/scoring calculations.
numpy
# NetworkX: models cross-repository dependency and relationship graphs.
networkx

# --- Approved structural code parsing (LRN-06) ---
# Tree-sitter parser runtime: used by the optional C# structural adapter.
tree-sitter==0.26.0
# Pinned C# grammar. No Roslyn or execution of untrusted source.
tree-sitter-c-sharp==0.23.5

# --- Google Cloud integration ---
# Cloud Storage: stores or retrieves approved application artifacts.
google-cloud-storage
# Secret Manager: retrieves service secrets when required.
google-cloud-secret-manager
# Cloud Logging: sends structured application logs to Google Cloud.
google-cloud-logging

# --- Reliability and observability ---
# Tenacity: retries transient failures such as temporary API errors.
tenacity
# Structlog: produces structured application logs.
structlog

# --- Local development and CI checks (also installed in container images) ---
# Pytest: runs automated Python unit and integration tests.
pytest
# Pytest AsyncIO: supports testing asynchronous Python functions.
pytest-asyncio
# Pytest Cov: reports test coverage for Python code.
pytest-cov
# Ruff: checks Python lint rules and formatting consistency.
ruff
